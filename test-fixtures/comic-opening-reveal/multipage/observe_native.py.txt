"""Observe the actual native renderer and encoder; never replace media work.

Usage: python3 comic-native-matrix-20260912.py RENDERER DIRECTORY EST COUNT [DURATION] [FPS]
Repeated retained art is a mechanical fixture, not newly generated story proof.
"""
import hashlib
import json
import os
from pathlib import Path
import runpy
import subprocess
import sys

from PIL import Image, ImageDraw
import numpy as np

repo = Path.cwd()
renderer, root, est, count = Path(sys.argv[1]), Path(sys.argv[2]), float(sys.argv[3]), int(sys.argv[4])
duration = float(sys.argv[5]) if len(sys.argv) > 5 else 2.3
fps = int(sys.argv[6]) if len(sys.argv) > 6 else 30
root.mkdir(parents=True, exist_ok=True)
source = repo / 'public/golden/comic/comic3d.jpg'
timeline = dict(out_w=1920, out_h=1080, fps=fps, est=est, per_page=4, turn=1.3,
                panels=[dict(panelIndex=i, img=str(source), dur=duration, avoid=[],
                             bubbles=[dict(id=f'cue-{i}', text='Hold on.', at=min(1.3, max(0, duration-.5)))]) for i in range(count)])
timeline_path = root / 'timeline.json'
timeline_path.write_text(json.dumps(timeline, indent=2))
output = root / 'native.mp4'
print(json.dumps(dict(root=str(root), renderer=str(renderer), est=est, duration=duration, fps=fps)), flush=True)
real_popen = subprocess.Popen
hashes, frame_meta, sampled, calls = [], [], [], []


def observe_frame(frame, event, arg):
    if event == 'return' and frame.f_code.co_name == 'render_page_frame':
        data = frame.f_locals
        calls.append(dict(t=data['t'], hand=data['hand_pt'], box=data['active']['box'] if data['active'] else None))


class ObservedInput:
    def __init__(self, stream):
        self.stream = stream

    def write(self, data):
        state = sys._getframe(1).f_globals
        number = len(hashes)
        hashes.append(hashlib.sha256(data).hexdigest())
        first = state['PAGES'][0]
        pane = first['panes'][0]
        x, y, w, h = pane['box']
        crop = first['world'][first['PY']+y:first['PY']+y+h, first['PX']+x:first['PX']+x+w]
        coverage = float(np.all(crop == pane['art'], axis=2).mean())
        frame_meta.append(dict(frame=number, t=state['t'], segment=state['seg'], openingPixelCoverage=coverage,
                               calls=[dict(t=c['t'], hand=list(map(float, c['hand'])) if c['hand'] else None) for c in calls]))
        calls.clear()
        if number % max(1, round(fps / 2)) == 0 or number == state['nframes'] - 1:
            image = Image.frombytes('RGB', (state['OW'], state['OH']), data)
            image.save(root / f'frame-{number:04d}.png')
            sampled.append(number)
        return self.stream.write(data)

    def close(self):
        return self.stream.close()


def observed_popen(args, *a, **kw):
    child = real_popen(args, *a, **kw)
    if args[0] == 'ffmpeg' and '-f' in args and 'rawvideo' in args:
        child.stdin = ObservedInput(child.stdin)
    return child


subprocess.Popen = observed_popen
sys.path.insert(0, str(repo / 'scripts'))
sys.argv = [str(renderer), str(timeline_path), str(root), str(output), str(repo / 'src/assets/whiteboard/hand.png')]
sys.setprofile(observe_frame)
try:
    state = runpy.run_path(str(renderer), run_name='__main__')
finally:
    sys.setprofile(None)
    subprocess.Popen = real_popen

assert state['ff'].returncode == 0, 'actual native encoder failed'
probe = json.loads(subprocess.check_output(['ffprobe', '-v', 'error', '-count_frames', '-select_streams', 'v:0',
    '-show_entries', 'stream=width,height,r_frame_rate,nb_read_frames:format=duration', '-of', 'json', str(output)]))
assert int(probe['streams'][0]['nb_read_frames']) == len(hashes)
sheet = Image.new('RGB', (4 * 480, ((len(sampled) + 3) // 4) * 292), '#17191d')
for i, number in enumerate(sampled):
    shot = Image.open(root / f'frame-{number:04d}.png').resize((480, 270))
    x, y = (i % 4) * 480, (i // 4) * 292
    sheet.paste(shot, (x, y))
    ImageDraw.Draw(sheet).text((x+8, y+273), f'{number/fps:.3f}s / frame {number}', fill='white')
sheet.save(root / 'contact-sheet.png')
receipt = dict(scope='Actual native frames/encoder. Repeated legacy comic poster; not channel/story/voice-quality proof.',
    renderer=str(renderer), rendererSha256=hashlib.sha256(renderer.read_bytes()).hexdigest(),
    sourceSha256=hashlib.sha256(source.read_bytes()).hexdigest(), outputSha256=hashlib.sha256(output.read_bytes()).hexdigest(),
    probe=probe, segments=state['segments'], sampledFrames=sampled, rawFrameSha256=hashes, observations=frame_meta,
    bubbleTimeline=json.loads((root / 'motion_comic_review_timeline.json').read_text()))
(root / 'evidence.json').write_text(json.dumps(receipt, indent=2))
print(json.dumps(dict(root=str(root), frames=len(hashes), samples=len(sampled), probe=probe)))
